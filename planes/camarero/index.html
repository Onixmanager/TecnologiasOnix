// api/helpers.js // Utilidades compartidas por las funciones serverless de Stripe. /** * Lee el body de la petición como Buffer crudo. * Necesario porque en Vercel Serverless Functions (no Next.js) no hay * bodyParser automático, y el webhook de Stripe necesita el body sin * procesar para verificar la firma. */ function getRawBody(req) { if (Buffer.isBuffer(req.rawBody)) return Promise.resolve(req.rawBody); if (typeof req.rawBody === 'string') return Promise.resolve(Buffer.from(req.rawBody, 'utf8')); if (Buffer.isBuffer(req.body)) return Promise.resolve(req.body); if (typeof req.body === 'string') return Promise.resolve(Buffer.from(req.body, 'utf8')); return new Promise((resolve, reject) => { const chunks = []; req.on('data', (chunk) => chunks.push(chunk)); req.on('end', () => resolve(Buffer.concat(chunks))); req.on('error', reject); }); } /** * Lee y parsea el body como JSON. Lanza un error con statusCode 400 si el * JSON es inválido. */ async function readJsonBody(req) { if (req.body && typeof req.body === 'object' && !Buffer.isBuffer(req.body)) return req.body; const raw = await getRawBody(req); if (!raw || raw.length === 0) return {}; try { return JSON.parse(raw.toString('utf8')); } catch (err) { const e = new Error('JSON inválido en el cuerpo de la petición'); e.statusCode = 400; throw e; } } /** * Verifica el ID token de Firebase enviado en el header * "Authorization: Bearer " y comprueba que el usuario autenticado * pertenece realmente al restaurantId que está pidiendo. Así evitamos que * un usuario pueda crear sesiones de Checkout/Portal para el restaurante * de otra persona. * * Lanza un error con `statusCode` (401/403/400) si algo falla. * Devuelve el decoded token de Firebase si todo es correcto. */ async function verificarUsuarioRestaurante(req, admin, restaurantId) { const authHeader = req.headers['authorization'] || ''; const idToken = authHeader.startsWith('Bearer ') ? authHeader.slice(7).trim() : null; if (!idToken) { const e = new Error('No autenticado: falta el token de Firebase'); e.statusCode = 401; throw e; } if (!restaurantId) { const e = new Error('restaurantId requerido'); e.statusCode = 400; throw e; } let decoded; try { decoded = await admin.auth().verifyIdToken(idToken); } catch (err) { const e = new Error('Token inválido o expirado'); e.statusCode = 401; throw e; } const snap = await admin.database().ref(`usuarios/${decoded.uid}/restaurantId`).get(); const miRestaurantId = snap.exists() ? snap.val() : null; if (!miRestaurantId || miRestaurantId !== restaurantId) { const e = new Error('No tienes permiso sobre este restaurante'); e.statusCode = 403; throw e; } return decoded; } // Orígenes permitidos a llamar a estos endpoints vía fetch() desde el // navegador. Incluye la propia app y la página externa de planes // (tecnologiasonix.com/planes/camarero/), que es la que ahora inicia el // checkout de Stripe cuando el usuario pulsa "pagar" en ella. const ALLOWED_CORS_ORIGINS = [ 'https://tecnologiasonix.com', 'https://www.tecnologiasonix.com', ]; /** * Aplica las cabeceras CORS necesarias para que la página externa de * planes pueda llamar a este endpoint desde el navegador del usuario, y * responde directamente a las peticiones preflight (OPTIONS). * * Devuelve `true` si ya se ha respondido a la petición (era un preflight * OPTIONS) y el handler que llama a esta función debe devolver * inmediatamente sin seguir procesando. Devuelve `false` si debe continuar * con normalidad (GET/POST reales). */ function aplicarCors(req, res) { const origin = req.headers['origin']; if (origin && ALLOWED_CORS_ORIGINS.includes(origin)) { res.setHeader('Access-Control-Allow-Origin', origin); res.setHeader('Vary', 'Origin'); } res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type, Authorization'); if (req.method === 'OPTIONS') { res.status(204).end(); return true; } return false; } module.exports = { getRawBody, readJsonBody, verificarUsuarioRestaurante, aplicarCors };
